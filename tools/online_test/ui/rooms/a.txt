# The owner: log in, create a room with a password, play from the title, look at the room, leave.
wait-title
sleep 4
panel open
screen login
fill login.name $UI_A_NAME
login $UI_A_NAME $UI_A_PASSWORD
expect-screen rooms
screen create
fill create.name $UI_ROOM
fill create.password $UI_ROOM_PASSWORD
fill create.players 6
fill create.camera 1
wait 10
shot create
create $UI_ROOM password=$UI_ROOM_PASSWORD max=6 camera=1 light=1
wait-session
sleep 2
shot online_title
menu online
wait-members 2 180
sleep 3
wait 10
shot room_two
sleep 20
fill leave.asked 1
wait 10
shot leave_question
leave
wait-left
sleep 2
wait 10
shot after_leave
end
