# The owner: log in, create a room for two with a password, the room view, Play, the room in F1, leave.
wait-title
to-menu
pad DOWN DOWN A
wait-screens
screen login
fill login.name $UI_A_NAME
wait 10
shot login
login $UI_A_NAME $UI_A_PASSWORD
expect-screen rooms
screen create
fill create.name $UI_ROOM
fill create.password $UI_ROOM_PASSWORD
fill create.players 2
fill create.camera 1
wait 10
shot create
create $UI_ROOM password=$UI_ROOM_PASSWORD max=2 camera=1 light=1
wait-session
sleep 2
wait 10
shot room_started
# Play (the room view's first button) closes the screens: the online title.
key pad-a
sleep 1
expect-screens closed
sleep 4
wait 10
shot online_title
menu online
wait-members 2 180
sleep 3
wait 10
shot room_two
sleep 45
fill leave.asked 1
wait 10
shot leave_question
leave
wait-left
sleep 2
wait 10
shot after_leave
end
