# The guest: the room list, the password prompt, a wrong password, joining.
wait-title
panel open
screen login
login $UI_B_NAME $UI_B_PASSWORD
expect-screen rooms
wait-room $UI_ROOM
select $UI_ROOM
wait 10
shot rooms
join $UI_ROOM
expect-screen join
wait 10
shot join_password
join $UI_ROOM password=not-it
expect-screen join
expect-error The room password is wrong.
wait 10
shot join_wrong
join $UI_ROOM password=$UI_ROOM_PASSWORD
wait-session
menu online
wait-members 2
sleep 3
wait 10
shot room_guest
sleep 40
end
